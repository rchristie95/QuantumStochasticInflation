"""Numerical acceptance checks for the examples shown in opening.py."""
import numpy as np
from oscillator_states import Q, DQ, DR, canonical_map, trajectories, fock_branches, gaussian_wave, pure_wigner
from opening import simulated_paths, massive_covariances, collective_map
from sections.joint_state import gaussian_marginal


def main():
    eye = np.eye(2)
    symplectic = np.block([[np.zeros((2,2)), eye], [-eye, np.zeros((2,2))]])
    for n in (1, 2, 5, 100):
        s = collective_map(n)
        a=s[:2,:2]; b=s[2:,2:]
        np.testing.assert_allclose(a@b.T,eye,atol=1e-14)
        np.testing.assert_allclose(s@symplectic@s.T,symplectic,atol=1e-14)
        # Average/average has commutator i/n; average/total has i.
        q_weights=np.ones(n)/n
        np.testing.assert_allclose(q_weights@(np.ones(n)/n),1/n)
        np.testing.assert_allclose(q_weights@np.ones(n),1)
        # Identical microscopic Gaussian inputs stay uncorrelated in this map.
        compatible=np.diag([1/n,1,n/4,1/4])
        mapped=s@compatible@s.T
        np.testing.assert_allclose(mapped[np.ix_([0,2],[1,3])],0,atol=1e-14)
        np.testing.assert_allclose(mapped[np.ix_([0,2],[0,2])],np.diag([1/(n+1),(n+1)/4]))
    c1 = np.diag([1,.25]); c2 = np.diag([.25,1])
    assert np.isclose(np.linalg.det(c1),.25)
    assert np.isclose(np.linalg.det(c2),.25)
    out=np.diag([5/16,5/4])
    assert np.isclose(1/(2*np.sqrt(np.linalg.det(out))),.8)
    for squeeze in np.linspace(0,np.log(2)/2,31):
        covariance=np.diag([.5*np.exp(2*squeeze),.5*np.exp(-2*squeeze)])
        assert np.isclose(np.linalg.det(covariance),.25)
    transform=collective_map(1)
    joint_cov=transform@np.diag([1,.25,.25,1])@transform.T
    np.testing.assert_allclose(joint_cov[np.ix_([0,2],[0,2])],out)
    assert np.isclose(np.linalg.det(joint_cov),1/16)  # pure two-mode output
    assert np.linalg.norm(joint_cov[np.ix_([0,2],[1,3])]) > 0
    # The displayed heatmaps must integrate to one and reproduce their covariance.
    grid=np.linspace(-10,10,801); dx=grid[1]-grid[0]
    xx,yy=np.meshgrid(grid,grid)
    sigma_in=np.diag([1,.25,.25,1])
    for fraction in (0,.5,1):
        mapping=canonical_map(1,fraction)
        full_cov=mapping@sigma_in@mapping.T
        for indices in ([0,1],[3,2],[0,2]):
            marginal_cov=full_cov[np.ix_(indices,indices)]
            density=gaussian_marginal(marginal_cov,xx,yy)
            np.testing.assert_allclose(density.sum()*dx**2,1,atol=1e-10)
            measured=np.array([[(density*xx**2).sum(),(density*xx*yy).sum()],
                               [(density*xx*yy).sum(),(density*yy**2).sum()]])*dx**2
            np.testing.assert_allclose(measured,marginal_cov,atol=1e-10)
        # W'_n(S z)=W_n(z): the relabelling has no symplectic Jacobian factor.
        old=np.array([.2,-.3,.4,-.5]); new=mapping@old
        original=np.exp(-old[0]**2/2-2*old[1]**2-2*old[2]**2-old[3]**2/2)/np.pi**2
        transformed=np.exp(-new@np.linalg.solve(full_cov,new)/2)/((2*np.pi)**2*np.sqrt(np.linalg.det(full_cov)))
        np.testing.assert_allclose(transformed,original,atol=1e-14)
    # Expanded joint expression and evaluated trace displayed on actions 41--43.
    rng=np.random.default_rng(41)
    for point in rng.normal(size=(20,4)):
        q,r,p,s=point
        expanded=np.exp(-2.5*q*q+1.5*q*r-.625*r*r-.625*p*p-1.5*p*s-2.5*s*s)/np.pi**2
        exact=np.exp(-point@np.linalg.solve(joint_cov,point)/2)/np.pi**2
        np.testing.assert_allclose(expanded,exact,rtol=1e-13,atol=1e-15)
    reduced_density=gaussian_marginal(out,xx,yy)
    explicit_trace=4/(5*np.pi)*np.exp(-8*xx**2/5-2*yy**2/5)
    np.testing.assert_allclose(reduced_density,explicit_trace,rtol=1e-13,atol=1e-15)
    np.testing.assert_allclose(2*np.pi*(reduced_density**2).sum()*dx**2,.8,atol=1e-10)
    np.testing.assert_allclose(gaussian_marginal(out,0,0),.8/np.pi,atol=1e-14)
    # Wavefunction in the new position coordinates has a nonzero QR coupling.
    position_cov=joint_cov[:2,:2]
    amplitude_matrix=np.linalg.inv(position_cov)/4
    np.testing.assert_allclose(amplitude_matrix,np.array([[5/4,-3/8],[-3/8,5/16]]))
    assert np.isclose(-2*amplitude_matrix[0,1],3/4)
    for n in (1,2,3,10):
        np.testing.assert_allclose(canonical_map(n),collective_map(n),atol=1e-14)
        assert not np.allclose(collective_map(n)[:2,:2]@collective_map(n)[:2,:2].T,eye)
        for fraction in np.linspace(0,1,11):
            path=canonical_map(n,fraction)
            np.testing.assert_allclose(path@symplectic@path.T,symplectic,atol=1e-13)
    amplitudes,probabilities,joint=fock_branches(gaussian_wave(Q,1),1)
    kernel=joint@joint.T*DR
    q,qp=np.meshgrid(Q,Q,indexing='ij')
    analytic=np.exp(-(q+qp)**2/(8*out[0,0])-out[1,1]*(q-qp)**2/2)/np.sqrt(2*np.pi*out[0,0])
    np.testing.assert_allclose(kernel,analytic,atol=2e-7)
    np.testing.assert_allclose(amplitudes@amplitudes.T,kernel,atol=2e-7)
    assert probabilities.sum()>1-1e-6
    assert np.isclose(np.trace(kernel)*DQ,1,atol=1e-7)
    assert np.isclose(np.sum(kernel**2)*DQ**2,.8,atol=1e-7)
    for row in trajectories():
        for kind in ('fock','position'):
            psi=row[kind]
            assert np.isclose(np.sum(np.abs(psi)**2)*DQ,1,atol=1e-7)
            momentum,wigner=pure_wigner(psi)
            dp=momentum[1]-momentum[0]
            assert np.isclose(wigner.sum()*DQ*dp,1,atol=1e-7)
            assert np.isclose(2*np.pi*np.sum(wigner**2)*DQ*dp,1,atol=2e-5)
        if row['n']>1:
            assert row['fock_completeness']>1-1e-5
            assert row['fock_probability']>0
        print(f"n={row['n']}: master purity={1/(2*np.sqrt(np.linalg.det(row['master']))):.6f}")
    second=trajectories()[1]
    momentum,wigner=pure_wigner(second['fock'])
    assert np.isclose(wigner[len(Q)//2,len(momentum)//2],-1/np.pi,atol=1e-6)
    expected_density=np.exp(-.8**2/(2*1.25))/np.sqrt(2*np.pi*1.25)
    assert np.isclose(second['position_density'],expected_density,atol=1e-7)
    distribution=np.abs(second['position'])**2
    mean=np.sum(Q*distribution)*DQ
    assert np.isclose(mean,.24,atol=1e-7)
    assert np.isclose(np.sum((Q-mean)**2*distribution)*DQ,.2,atol=1e-7)
    t,paths,dw=simulated_paths()
    t2,paths2,dw2=simulated_paths()
    np.testing.assert_array_equal(paths,paths2)
    np.testing.assert_allclose(np.diff(paths,axis=1),-.03*paths[:,:-1]*(t[1]-t[0])+dw)
    assert np.isfinite(paths).all()
    assert np.max(np.abs(paths)) < 7, "Displayed paths and histogram must fit the axes"
    # Discrete normalized smearing: [Q, Pi_bar]/i = sum w_j^2/dV.
    weights=np.array([.1,.2,.3,.4]); cell_volume=.7
    inverse_volume=np.sum(weights**2)/cell_volume
    effective_volume=1/inverse_volume
    assert not np.isclose(inverse_volume,1)
    assert np.isclose(effective_volume*inverse_volume,1)
    for mass in (.5,1.5,3.):
        t,covs,purities,stationary,drift,diffusion,phi,pi=massive_covariances(mass)
        np.testing.assert_allclose(phi*np.conj(pi)-np.conj(phi)*pi,1j,atol=1e-12)
        np.testing.assert_allclose(drift@stationary+stationary@drift.T+diffusion,0,atol=1e-10)
        assert np.isclose(np.linalg.det(diffusion),9/4)
        volume=6*np.pi**2/.1**3
        analytic=np.array([[(abs(pi)**2+6*np.real(np.conj(phi)*pi)+(9+mass**2)*abs(phi)**2)/(2*mass**2*volume),
                             -1.5*abs(phi)**2],[-1.5*abs(phi)**2,volume*(abs(pi)**2+mass**2*abs(phi)**2)/2]])
        scale=np.diag([np.sqrt(mass*volume),1/np.sqrt(mass*volume)])
        np.testing.assert_allclose(stationary,scale@analytic@scale.T,rtol=1e-11,atol=1e-11)
        assert np.min(np.linalg.eigvalsh(covs))>0
        assert np.all(np.linalg.det(covs)>=.25-1e-10)
        assert np.all(purities<=1+1e-10)
        assert np.isclose(purities[0],1)
        print(f"m/H={mass:g}: stationary purity={1/(2*np.sqrt(np.linalg.det(stationary))):.6f}")
    print("PASS: average/total canonical normalization, symplectic map, Fock/position completeness and Wigner functions, trajectories, massive-field covariance")


if __name__ == "__main__":
    main()
